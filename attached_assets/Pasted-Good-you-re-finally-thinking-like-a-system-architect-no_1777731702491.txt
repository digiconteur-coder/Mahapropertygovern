server/index.js

const express = require("express");
const http = require("http");
const socketIo = require("socket.io");
const app = express();
const server = http.createServer(app);
const io = socketIo(server, { cors: { origin: "*" } });
app.use(express.json());
app.use("/property", require("./routes/property"));
app.use("/developer", require("./routes/developer"));
io.on("connection", (socket) => {
  console.log("User connected");
  socket.on("updateProperty", (data) => {
    io.emit("propertyUpdated", data);
  });
});
server.listen(5000, () => console.log("Server running"));

import { useEffect, useState } from "react";
import io from "socket.io-client";
const socket = io("http://localhost:5000");
export default function Home() {
  const [updates, setUpdates] = useState([]);
  useEffect(() => {
    socket.on("propertyUpdated", (data) => {
      setUpdates((prev) => [data, ...prev]);
    });
  }, []);
  return (
    <div className="p-4">
      
      {/* HEADER */}
      <div className="flex justify-between items-center">
        <h1 className="text-xl font-bold">Bharat Property Card</h1>
        <img src="/profile.png" className="w-10 h-10 rounded-full"/>
      </div>
      {/* GOVT BANNER */}
      <div className="mt-4 bg-blue-500 text-white p-3 rounded">
        Govt Scheme: Digital Property Drive Active
      </div>
      {/* MAIN ACTION */}
      <div className="mt-6 grid grid-cols-2 gap-4">
        <button className="bg-green-500 p-4 rounded">View Properties</button>
        <button className="bg-purple-500 p-4 rounded">Scan QR</button>
      </div>
      {/* LIVE UPDATES */}
      <div className="mt-6">
        <h2 className="font-bold">Live Updates</h2>
        {updates.map((u, i) => (
          <div key={i} className="bg-gray-100 p-2 mt-2 rounded">
            {u.message}
          </div>
        ))}
      </div>
    </div>
  );
}
